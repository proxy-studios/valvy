using UnityEngine;

public class NetworkManager : MonoBehaviour
{
    public static NetworkManager Instance { get; private set; }

    [Header("Server Config")]
    public int port = 7777;
    public string serverIP = "127.0.0.1";

    [Header("Status")]
    public bool isServer;
    public bool isClient;
    public int localClientId = -1;

    public ValvyServer Server { get; private set; }
    public ValvyClient Client { get; private set; }
    public PlayerNetworkSync PlayerSync { get; private set; }

    private void Awake()
    {
        if (Instance != null && Instance != this)
        {
            Destroy(gameObject);
            return;
        }
        Instance = this;
        DontDestroyOnLoad(gameObject);

        Server = gameObject.AddComponent<ValvyServer>();
        Client = gameObject.AddComponent<ValvyClient>();
        PlayerSync = gameObject.AddComponent<PlayerNetworkSync>();
    }

    private void Start()
    {
        AutoConnect();
    }

    public void AutoConnect()
    {
        if (!Client.ConnectToServer(serverIP, port))
        {
            Debug.Log("[Valvy] No host found. Starting local Host...");
            StartHost();
        }
        else
        {
            isClient = true;
        }
    }

    public void StartHost()
    {
        Server.StartServer(port);
        isServer = true;

        if (Client.ConnectToServer(serverIP, port))
        {
            isClient = true;
        }
    }

    public void Disconnect()
    {
        if (Server != null) Server.StopServer();
        if (Client != null) Client.DisconnectClient();
        if (PlayerSync != null) PlayerSync.ClearPlayers();

        isServer = false;
        isClient = false;
        localClientId = -1;
    }

    private void OnApplicationQuit()
    {
        Disconnect();
    }
}

Why Choose Valvy?

⚡

Smart AutoConnect

Automatically scans for an active host on launch. If no server responds, Valvy dynamically spins up a local host server.

🔒

Direct Raw Sockets

Built straight on System.Net.Sockets for maximum control, zero third-party dependencies, and minimal overhead.

🕶️

PlayerNetworkSync Built-in

Modular components attach dynamically on Awake to manage player states and clear local entities seamlessly on disconnect.

📦

Tiny Footprint

The entire setup runs through lightweight MonoBehaviours without heavy inspector configurations or external packages.

🛠️

Single Pattern Core

Built around a singleton NetworkManager instance that persists cleanly across scenes with DontDestroyOnLoad.

🌐

Cross-Platform

Runs seamlessly across Windows desktop, Standalone VR devices, and mobile C# environments.

NetworkManager API

NetworkManager.Instance.AutoConnect()

Attempts to join target server IP. Automatically executes StartHost() if connection fails.

NetworkManager.Instance.StartHost()

Initializes the local ValvyServer and connects the local ValvyClient to localhost on port 7777.

NetworkManager.Instance.Disconnect()

Stops server, disconnects client, clears synced players, and resets network status flags.

PlayerNetworkSync

Dynamically attached component handling synchronized player transform dictionaries and lifecycle cleanup.
